单轴行驶与原地转向

真实源码中的轨迹段规划和采样。把连续行驶、停车和原地转向表示为不同状态，不带任何现场路径坐标。

function axisTrack(initial,steps,initialWheel=0){
  let from=initial, wheel=initialWheel;
  const segments=steps.map(([range,x,z,yaw])=>{
    const [start,end]=range, to=pose(x,0,z,yaw), dx=x-from.x, dz=z-from.z;
    const length=Math.hypot(dx,dz), angle=yaw-from.yaw;
    if(end<=start||(dx&&dz)||(length&&angle)||(!length&&Math.abs(Math.abs(angle)-PI/2)>1e-9)){
      throw new Error('CTU 轨迹段应为单轴直行或原地转 90°');
    }
    const curve=length?new THREE.LineCurve3(V(from.x,0,from.z),V(x,0,z)):null;
    const signed=length*Math.sign(-dx*Math.sin(yaw)-dz*Math.cos(yaw));
    const step={start,end,from,to,length,curve,wheel,signed,kind:length?'straight':'pivot'};
    from=to;
    wheel+=signed;
    return step;
  });
  return {segments,endWheel:wheel,sample(t){
    const step=segments.find(s=>t<s.end);
    if(!step)return {body:{...from},wheel,turning:false,phase:'hold'};
    const {start,end,from:origin,to,curve,signed}=step;
    if(t<start)return {body:{...origin},wheel:step.wheel,turning:false,phase:'hold'};
    if(!curve)return {body:pose(origin.x,0,origin.z,lerp(origin.yaw,to.yaw,linearProgress(t,[start,end]))),
      wheel:step.wheel,turning:true,phase:'pivot'};
    const q=rigidProgress(t,[start,end]), p=curve.getPoint(q);
    return {body:pose(p.x,0,p.z,origin.yaw),wheel:step.wheel+signed*q,turning:false,phase:'straight'};
  }};
}
