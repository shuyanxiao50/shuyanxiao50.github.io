import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

plt.figure(figsize=(10, 6))

plt.hist(df['salary_usd'], bins=20, color='skyblue', edgecolor='black')

plt.title('AI Job Salary Distribution')
plt.xlabel('Salary (USD)')
plt.ylabel('Count')

plt.show()

国家平均薪资比较

import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

top_countries = df.groupby('country')['salary_usd'].mean().sort_values(ascending=False).head(10)

plt.figure(figsize=(12, 6))
bars = plt.bar(top_countries.index, top_countries.values, color='teal')

plt.title('Top 10 Countries by Average AI Salary', fontsize=14)
plt.xlabel('Country')
plt.ylabel('Average Salary (USD)')
plt.xticks(rotation=45)

for bar in bars:
    height = bar.get_height()
    plt.text(bar.get_x() + bar.get_width()/2., height, f'${int(height)}', ha='center', va='bottom')

plt.show()

import seaborn as sns

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

plt.figure(figsize=(10, 6))

数值特征相关性

import seaborn as sns
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

numeric_cols = ['experience_years', 'salary_usd', 'weekly_hours', 'company_rating', 'job_openings']

corr_matrix = df[numeric_cols].corr()

plt.figure(figsize=(10, 8))
sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', fmt=".2f", linewidths=0.5)

plt.title('Correlation Heatmap of Key Features', fontsize=14)
plt.show()

经验年数与薪资关系

import seaborn as sns
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

plt.figure(figsize=(12, 7))

sns.scatterplot(data=df, x='experience_years', y='salary_usd',
                hue='experience_level', alpha=0.6, palette='deep')

plt.title('Salary vs Experience Years (Grouped by Level)', fontsize=14)
plt.xlabel('Experience Years')
plt.ylabel('Salary (USD)')
plt.legend(title='Experience Level')
plt.show()

企业规模分组分布

import seaborn as sns
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

plt.figure(figsize=(12, 7))

sns.kdeplot(data=df, x='salary_usd', hue='company_size', fill=True, common_norm=False, palette='viridis', alpha=0.5)

plt.title('Salary Distribution Comparison by Company Size', fontsize=14)
plt.xlabel('Salary (USD)')
plt.ylabel('Density')

plt.xlim(0, 200000)
plt.show()
